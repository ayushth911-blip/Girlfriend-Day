            // Fireflies (Background Canvas)
            const bgCanvas = document.getElementById('bg-canvas');
            const bgCtx = bgCanvas.getContext('2d');
            let w, h, fireflies = [];

            function resizeBg() {
                w = bgCanvas.width = window.innerWidth;
                h = bgCanvas.height = window.innerHeight;
            }
            window.addEventListener('resize', resizeBg);
            resizeBg();

            class Firefly {
                constructor() {
                    this.x = Math.random() * w;
                    this.y = Math.random() * h;
                    this.s = Math.random() * 2 + 1;
                    this.vx = (Math.random() - 0.5) * 0.5;
                    this.vy = (Math.random() - 0.5) * 0.5;
                    this.alpha = Math.random();
                    this.alphaChange = (Math.random() * 0.02) + 0.005;
                }
                update() {
                    this.x += this.vx;
                    this.y += this.vy;
                    if(this.x < 0 || this.x > w) this.vx *= -1;
                    if(this.y < 0 || this.y > h) this.vy *= -1;
                    this.alpha += this.alphaChange;
                    if (this.alpha <= 0.1 || this.alpha >= 1) this.alphaChange *= -1;
                }
                draw() {
                    bgCtx.beginPath();
                    bgCtx.arc(this.x, this.y, this.s, 0, Math.PI * 2);
                    bgCtx.fillStyle = `rgba(255, 75, 114, ${this.alpha})`;
                    bgCtx.shadowBlur = 10;
                    bgCtx.shadowColor = '#ff4b72';
                    bgCtx.fill();
                }
            }

            for(let i=0; i<50; i++) fireflies.push(new Firefly());

            function animBg() {
                bgCtx.clearRect(0, 0, w, h);
                fireflies.forEach(f => { f.update(); f.draw(); });

            // Fireworks (Foreground Canvas) on Surprise Button
            const fxCanvas = document.getElementById('fx-canvas');
            const fxCtx = fxCanvas.getContext('2d');
            let fw, fh, particles = [];
            
            function resizeFx() {
                fw = fxCanvas.width = window.innerWidth;
                fh = fxCanvas.height = window.innerHeight;
            }
            window.addEventListener('resize', resizeFx);
            resizeFx();

            class Particle {
                constructor(x, y, color) {
                    this.x = x; this.y = y;
                    this.color = color;
                    this.radius = Math.random() * 3 + 1;
                    const angle = Math.random() * Math.PI * 2;
                    const speed = Math.random() * 8 + 2;
                    this.vx = Math.cos(angle) * speed;
                    this.vy = Math.sin(angle) * speed;
                    this.friction = 0.95;
                    this.gravity = 0.1;
                    this.alpha = 1;
                    this.decay = Math.random() * 0.015 + 0.01;
                }
                update() {
                    this.vx *= this.friction;
                    this.vy *= this.friction;
                    this.vy += this.gravity;
                    this.x += this.vx;
                    this.y += this.vy;
                    this.alpha -= this.decay;
                }
                draw() {
                    fxCtx.save();
                    fxCtx.globalAlpha = this.alpha;
                    fxCtx.beginPath();
                    fxCtx.arc(this.x, this.y, this.radius, 0, Math.PI*2);
                    fxCtx.fillStyle = this.color;
                    fxCtx.shadowBlur = 10;
                    fxCtx.shadowColor = this.color;
                    fxCtx.fill();
                    fxCtx.restore();
                }
            }

            let fwActive = false;
            function animFx() {
                if(!fwActive) return;
                fxCtx.clearRect(0, 0, fw, fh);
                particles = particles.filter(p => p.alpha > 0);
                particles.forEach(p => { p.update(); p.draw(); });
                if(particles.length > 0) {
                    requestAnimationFrame(animFx);
                } else {
                    fwActive = false;
                    fxCtx.clearRect(0, 0, fw, fh);
                }
            }

            function explode(x, y) {
                const colors = ['#ff4b72', '#ff758c', '#ffffff', '#ffd700'];
                for(let i=0; i<100; i++) {
                    particles.push(new Particle(x, y, colors[Math.floor(Math.random() * colors.length)]));
                }
                if(!fwActive) {
                    fwActive = true;
                    animFx();
                }
            }

            const surpriseBtn = document.getElementById('surprise-btn');

                const rect = surpriseBtn.getBoundingClientRect();
                explode(rect.left + rect.width/2, rect.top);
                explode(rect.left, rect.top - 100);

                }
